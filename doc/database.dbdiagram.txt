// ============================================================
// POD Platform — ERD v1.3  (dbdiagram.io / DBML)
// Back to basics: clarity over cleverness.
// ============================================================
//
// Changes vs v1.2:
//   1. UN-MERGED provider_profiles back into two separate tables:
//      designer_profiles + printer_provider_profiles (v1.1 style).
//      The single-table-with-type-flag version was confusing, not simpler.
//   2. REMOVED THE ENTIRE CUSTOM-ORDER / NEGOTIATION FEATURE:
//      - customization_requests, custom_order_variants, custom_order_items
//      - the messages / thread_log idea
//      Custom requests are a future feature, not a v1 schema concern.
//   3. REMOVED sub_orders AND fulfillment_log AS FORMAL ENTITIES:
//      - printer_provider_id and status now live directly on order_items
//      - no more "batch of items going to the same printer" as its own row
//      - shipments reference order_id + printer_provider_id directly
//        (a logical grouping, not an enforced one)
//
// TRADE-OFF TO KNOW ABOUT: if two order_items in the same order go to the
// same printer, there's no shared entity representing that batch anymore.
// Each item's status updates independently. Fine for v1 as long as you
// don't need one atomic "this whole batch moved to printing" update.
//
// Result: 20 platform tables, and — more importantly — no confusing merges.
// Every table maps to one real-world thing.
//
// v1.3 enhancements layered on top of the v1.2 shape:
//   - Soft-delete added to product_templates, product_variants,
//     design_product_mappings, order_items (joining users/designs/orders/payments)
//   - UUID columns (Pattern B) added to the 7 public-facing tables:
//     users, designs, design_product_mappings, product_templates,
//     product_variants, orders, payments. Integer id stays as PK + FK type.
//   - cart_items.variant_key STORED generated column replaces NULL-distinct
//     issue in the unique constraint
//   - Explicit indexes on FK columns and query-hot status/deleted_at fields
// ============================================================

// ===== AUTH =====
Table users {
  id bigint [pk, increment]
  uuid varchar [unique, note: 'public-facing UUID (Pattern B); used in URLs and external responses. Internal joins use id.']
  name varchar
  email varchar
  password varchar
  phone varchar
  role varchar [note: 'ENUM: customer, designer, printer_provider, admin — fixed, closed set']
  created_at timestamp
  updated_at timestamp
  deleted_at timestamp

  Indexes {
    uuid [unique, name: 'users_uuid_unique']
  }
}

// ===== PROFILES (separate again — clearer than a merged type-flag table) =====
Table designer_profiles {
  id bigint [pk, increment]
  user_id bigint [ref: > users.id, note: 'FK -> users.id']
  bio text
}

Table printer_provider_profiles {
  id bigint [pk, increment]
  user_id bigint [ref: > users.id, note: 'FK -> users.id']
  company_name varchar
}

Table addresses {
  id bigint [pk, increment]
  user_id bigint [ref: > users.id, note: 'FK -> users.id']
  line1 varchar
  city varchar
  country varchar
  phone varchar
}

// ===== CATALOG / DESIGN =====
Table categories {
  id bigint [pk, increment]
  name varchar
  parent_id bigint [ref: > categories.id, note: 'nullable, self-reference']

  Indexes {
    parent_id [name: 'idx_categories_parent']
  }
}

Table tags {
  id bigint [pk, increment]
  name varchar
}

Table design_tag {
  design_id bigint [ref: > designs.id, note: 'FK -> designs.id']
  tag_id bigint [ref: > tags.id, note: 'FK -> tags.id']

  Indexes {
    (design_id, tag_id) [unique, name: 'design_tag_pk']
  }
}

Table product_templates {
  id bigint [pk, increment]
  uuid varchar [unique, note: 'public-facing UUID (Pattern B); used in URLs']
  printer_provider_id bigint [ref: > printer_provider_profiles.id, note: 'FK']
  name varchar
  type varchar [note: 'free text / printer-managed, not a fixed enum']
  base_cost decimal
  specs json
  created_at timestamp
  updated_at timestamp
  deleted_at timestamp [note: 'soft-delete']

  Indexes {
    uuid [unique, name: 'product_templates_uuid_unique']
    printer_provider_id [name: 'product_templates_printer_provider_id_index']
  }
}

Table product_variants {
  id bigint [pk, increment]
  uuid varchar [unique, note: 'public-facing UUID (Pattern B); used in URLs']
  product_template_id bigint [ref: > product_templates.id, note: 'FK']
  attributes json [note: 'e.g. size, color, material']
  price_delta decimal
  sku varchar [note: 'nullable']
  is_active boolean
  created_at timestamp
  updated_at timestamp
  deleted_at timestamp [note: 'soft-delete']

  Indexes {
    uuid [unique, name: 'product_variants_uuid_unique']
    product_template_id [name: 'product_variants_product_template_id_index']
  }
}

Table designs {
  id bigint [pk, increment]
  uuid varchar [unique, note: 'public-facing UUID (Pattern B); used in URLs']
  designer_id bigint [ref: > designer_profiles.id, note: 'FK']
  category_id bigint [ref: > categories.id, note: 'FK']
  title varchar
  status varchar [note: 'ENUM: draft, published, archived — fixed, closed set']
  created_at timestamp
  updated_at timestamp
  deleted_at timestamp [note: 'soft-delete']

  Indexes {
    uuid [unique, name: 'designs_uuid_unique']
    designer_id [name: 'designs_designer_id_index']
    category_id [name: 'designs_category_id_index']
    status [name: 'designs_status_index']
    deleted_at [name: 'designs_deleted_at_index']
  }
}

Table design_product_mappings {
  id bigint [pk, increment]
  uuid varchar [unique, note: 'public-facing UUID (Pattern B); used in URLs']
  design_id bigint [ref: > designs.id, note: 'FK']
  product_template_id bigint [ref: > product_templates.id, note: 'FK']
  preferred_printer_id bigint [ref: > printer_provider_profiles.id, note: 'FK']
  final_price decimal
  created_at timestamp
  updated_at timestamp
  deleted_at timestamp [note: 'soft-delete']

  Indexes {
    uuid [unique, name: 'design_product_mappings_uuid_unique']
    (design_id, product_template_id) [unique, name: 'dpm_design_template_unique']
    design_id [name: 'design_product_mappings_design_id_index']
    product_template_id [name: 'design_product_mappings_product_template_id_index']
    preferred_printer_id [name: 'design_product_mappings_preferred_printer_id_index']
  }
}

// ===== CART (persistent) =====
Table cart_items {
  id bigint [pk, increment]
  user_id bigint [ref: > users.id, note: 'FK']
  design_product_mapping_id bigint [ref: > design_product_mappings.id, note: 'FK']
  product_variant_id bigint [ref: > product_variants.id, note: 'FK — nullable, size/color/material choice']
  variant_key bigint [note: 'STORED generated: COALESCE(product_variant_id, 0). Coalesces NULL variant to 0 so the unique constraint catches duplicates when no variant is selected (SQLite/MySQL treat NULL as distinct in unique indexes).']
  quantity int
  created_at timestamp
  updated_at timestamp

  Indexes {
    (user_id, design_product_mapping_id, variant_key) [unique, name: 'cart_items_unique_line']
    user_id [name: 'cart_items_user_id_index']
  }
}

// ===== ORDERS / FULFILLMENT =====
Table orders {
  id bigint [pk, increment]
  uuid varchar [unique, note: 'public-facing UUID (Pattern B); used in order-tracking URLs']
  customer_id bigint [ref: > users.id, note: 'FK']
  shipping_address_id bigint [ref: > addresses.id, note: 'FK — reference only, not source of truth post-checkout']
  shipping_line1 varchar [note: 'snapshot at checkout time']
  shipping_city varchar [note: 'snapshot at checkout time']
  shipping_country varchar [note: 'snapshot at checkout time']
  shipping_phone varchar [note: 'snapshot at checkout time']
  status varchar [note: 'ENUM: pending, paid, processing, shipped, delivered, cancelled — fixed, closed set']
  total_amount decimal
  created_at timestamp
  updated_at timestamp
  deleted_at timestamp [note: 'soft-delete']

  Indexes {
    uuid [unique, name: 'orders_uuid_unique']
    customer_id [name: 'orders_customer_id_index']
    status [name: 'orders_status_index']
    created_at [name: 'orders_created_at_index']
  }
}

Table order_items {
  id bigint [pk, increment]
  order_id bigint [ref: > orders.id, note: 'FK']
  design_product_mapping_id bigint [ref: > design_product_mappings.id, note: 'FK']
  product_variant_id bigint [ref: > product_variants.id, note: 'FK — nullable']
  printer_provider_id bigint [ref: > printer_provider_profiles.id, note: 'FK — which printer fulfills THIS item. Replaces sub_orders: no separate batch entity, each item tracked independently']
  status varchar [note: 'ENUM: pending, received, printing, printed, handed_off, cancelled — fixed, closed set. Per-item fulfillment status, replaces sub_orders.status']
  quantity int
  unit_price decimal
  created_at timestamp
  updated_at timestamp
  deleted_at timestamp [note: 'soft-delete — parallels orders.softDeletes so order restoration brings items back.']

  Indexes {
    printer_provider_id [name: 'order_items_printer_provider_id_index']
    status [name: 'order_items_status_index']
  }
}

Table delivery_companies {
  id bigint [pk, increment]
  name varchar
  coverage_zones json
  tracking_url_pattern varchar
}

Table shipments {
  id bigint [pk, increment]
  order_id bigint [ref: > orders.id, note: 'FK — one order can have multiple shipments if items came from different printers']
  printer_provider_id bigint [ref: > printer_provider_profiles.id, note: 'FK — identifies which printer\'s items this shipment covers (logical grouping, not an enforced junction to order_items)']
  delivery_company_id bigint [ref: > delivery_companies.id, note: 'FK']
  tracking_number varchar
  status varchar [note: 'ENUM: pending, shipped, delivered, returned — fixed, closed set']
  shipped_at timestamp
  delivered_at timestamp
  created_at timestamp
  updated_at timestamp

  Indexes {
    printer_provider_id [name: 'shipments_printer_provider_id_index']
    delivery_company_id [name: 'shipments_delivery_company_id_index']
  }
}

Table payments {
  id bigint [pk, increment]
  uuid varchar [unique, note: 'public-facing UUID (Pattern B); used in transaction references']
  order_id bigint [ref: > orders.id, note: 'FK']
  method varchar [note: 'ENUM: cash_on_delivery, bank_transfer, card — fixed for now']
  status varchar [note: 'ENUM: pending, confirmed, rejected — fixed, closed set']
  confirmed_by_admin_id bigint [ref: > users.id, note: 'FK — nullable']
  confirmed_at timestamp
  created_at timestamp
  deleted_at timestamp [note: 'soft-delete']
  Note: 'proof-of-payment image lives in MEDIA (model_type=Payment, collection_name=payment_proof)'

  Indexes {
    uuid [unique, name: 'payments_uuid_unique']
    method [name: 'payments_method_index']
    status [name: 'payments_status_index']
  }
}

// ===== SHARED INFRASTRUCTURE =====
Table media {
  id bigint [pk, increment]
  model_type varchar [note: 'polymorphic owner']
  model_id bigint
  collection_name varchar [note: 'ENUM: mockup, print_file, payment_proof, attachment — fixed, closed set']
  file_path varchar
}

Table notifications {
  id bigint [pk, increment]
  notifiable_type varchar
  notifiable_id bigint
  type varchar [note: 'NOT an enum — new notification types get added as features ship']
  data json
  read_at timestamp
}

Table settings {
  id bigint [pk, increment]
  key varchar
  value text
}
